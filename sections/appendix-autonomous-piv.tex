\section{High-Speed Autonomous PIV Data Templates}

This appendix summarizes the data templates used by the SIA High-Speed Autonomous PIV flow. Bracketed values (for example \texttt{[010203]}) denote illustrative placeholders pulled from device identifiers or credential contents. Current PIV Auto challenge generation uses a fixed-width KMAC256 input template rather than the earlier ASN.1 CMAC template.

\begingroup\footnotesize

\subsection{Fixed-Width KMAC Challenge Template}
\begin{verbatim}
4F 53 44 50 2D 50 49 56 2D 41 55 54 4F 01  # "OSDP-PIV-AUTO", version 1
[07000000]                                  # Series Counter, 32-bit LSB first
[2A000000]                                  # Sequence Counter, 32-bit LSB first
[D13810D828AB6C10C339E5A1685A08C92ADE0A6184E739C3E7] # FASC-N
[09D49C7EFDD0432EACEA268AE905274C]         # UUID, or 16 zero bytes
[9E]                                        # PIV key reference
[07]                                        # PIV algorithm identifier
[00112233445566778899AABBCCDDEEFF...]       # 32-byte supplemental entropy
\end{verbatim}

The challenge is \texttt{KMAC256(SC2-derived-key, template, 32, "OSDP-PIV-AUTO-CHALLENGE-v1")}. The supplemental entropy field is always present and fixed at 32 bytes; all zeroes are valid but not recommended.

\subsection{PIV Signing Request}
\begin{verbatim}
30 22                         # Sequence of length 0x22
06 0B 2B0601040182CC230A0101  # OID: SIA PIV Signing Request
A0 13                         # Explicit Tag: 0x00
03 21 00[...]                 # 32-byte KMAC256 PIV Auto challenge
\end{verbatim}

\subsection{Dynamic Authentication Templates}
\paragraph{RSA-1024}
\begin{verbatim}
7C 81 85                      # Dynamic Authentication Template
82 00                         # Explicit Tag: 0x02 (Request for Response)
81 81 80                      # 1024-bit (0x80) challenge
00                            # Begin padding
01                            # Signature operation, private key operation
FF ... FF                     # Pad with 0xFF to total length 0x80
00                            # End sentinel
30 2F
30 0B                         # OID: Algorithm (SHA-256)
06 09 608648016503040201
04 20 [000102...1E1F]         # Digest (OCTET STRING)
\end{verbatim}

\paragraph{RSA-2048}
\begin{verbatim}
7C 82 0106                    # Dynamic Authentication Template
82 00                         # Explicit Tag: 0x02 (Request for Response)
81 82 0100                    # 2048-bit (0x100) challenge
00                            # Begin padding
01                            # Signature operation, private key operation
FF ... FF                     # Pad with 0xFF to total length 0x100
00                            # End sentinel
30 2F
30 0B                         # OID: Algorithm (SHA-256)
06 09 608648016503040201
04 20 [000102...1E1F]         # Digest (OCTET STRING)
\end{verbatim}

\paragraph{EC}
\begin{verbatim}
7C 24                         # Dynamic Authentication Template
82 00                         # Explicit Tag: 0x02 (Request for Response)
81 20                         # 256-bit (0x20) digest
[000102...1E1F]
\end{verbatim}

\subsection{APDUs}
\paragraph{Select PIV Application}
\begin{verbatim}
- 00 A4 04 00  [09] A0 00 00 03 08 00 00 10 00  # Select PIV Application
< 61 11 4F 06 00 00 10 00 01 00 79 07 4F 05 A0 00 00 03 08 (90 00)
\end{verbatim}

\paragraph{Get Data (Discovery Object)}
\begin{verbatim}
- 00 CB 3F FF  [00 00 03]  5C 01 7E              # Get Data, tag 5C, object 0x7E
< 7E 12 [ 4F 0B [A0 00 00 03 08 00 00 10 00 01 00] [ 5F 2F 02 [40 00] ] ] (90 00)
\end{verbatim}

\paragraph{Verify PIN (contact-only)}
\begin{verbatim}
- 00 20 00 [00/80]  08  30 30 30 30 30 30 FF FF  # PIN padded to 8 octets with 0xFF
< (90 00)                                        # 0x63CX = fail (X retries left), 0x6983 = PIN blocked
\end{verbatim}

\paragraph{Sign (RSA-1024)}
\begin{verbatim}
- 00 87 [06] [9A/9E] 88 [DAT]  # 0x06 = RSA-1024; 0x9A = PIV Auth; 0x9E = Card Auth
< 7C 81 83 82 81 81 [signature x 0x10] (90 00)
\end{verbatim}

\paragraph{Sign (RSA-2048)}
\begin{verbatim}
- 10 87 [07] [9A/9E] D9 [DAT header and padding]   # 0x07 = RSA-2048
- 00 87 [07] [9A/9E] 31 [DAT encoded digest]
< 7C 82 01 04 82 82 01 00 [signature x 0xF8] (90 00) # First 0xF8 octets
- 00 C0 00 00
< [signature x 0x08] (90 00)                        # Remaining 0x08 octets
\end{verbatim}

\paragraph{Sign (EC)}
\begin{verbatim}
- 00 87 [06] [11/14] 26 [DAT]  # 0x11 = EC-P256; 0x14 = EC-P384; 0x9A = PIV Auth; 0x9E = Card Auth
< 7C [LEN] 82 [LEN] [signature, up to length 0x48] (90 00)
\end{verbatim}
