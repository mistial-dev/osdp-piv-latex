\begin{PacketTable}{Card Status Reply (osdp\_CARDSTATUSR)}
  \PacketRow{0x01}{CMND}{Reply identifier.}{0x85 (tentative)}
  \PacketRow{0x01}{Status Type}{Standard credential status, including successful pairing completion.}{0x01}
  \PacketRow{0x01}{Detected Credential Type}{Current card interface: \code{0x00} none, \code{0x01} ISO~7816 contact, \code{0x02} ISO~14443 contactless, \code{0x03} other ISO~7816 credential (for example DESFire, Seos), \code{0x04} low-frequency credential, \code{0x05} unknown type, \code{0x06}-\code{0x7F} reserved, \code{0x80}-\code{0xFF} private use.}{0x00-0xFF}
  \PacketRow{0x01}{Credential Security Status}{\code{0x00} SM absent; \code{0x01} SM established and VCI available; \code{0x02} SM established without VCI. Contact SM uses \code{0x02}. Values \code{0x03}-\code{0x7F} reserved, \code{0x80}-\code{0xFF} private use.}{0x00-0xFF}
  \PacketRow{0x01}{SM Trust Anchor ID}{Identifier assigned in \code{osdp_PIVVCILOADTA} for the anchor used to validate credential SM. The anchor's inventory IIN matches \emph{SM Anchor IIN}. Set to \code{0x00} when SM is absent.}{0x00-0xFF}
  \PacketRow{0x08}{SM CVC IIN}{Issuer Identification Number from the credential SM CVC tag \code{0x42}. Set to all zero when SM is absent.}{0x00-0xFF}
  \PacketRow{0x08}{SM Anchor IIN}{Issuer Identification Number of the loaded trust anchor used to validate credential SM. Set to all zero when SM is absent.}{0x00-0xFF}
  \PacketRow{0x02}{PIN Usage Policy}{PIN Usage Policy bytes (tag \code{0x5F2F}) returned verbatim when available. A value of \code{0xFFFF} indicates the policy is not present or has not yet been read.}{0x0000-0xFFFF}
  \PacketRow{0x01}{Selected AID Length}{Length of the selected AID. \code{0x00} indicates no application is currently selected.}{0x00-0x10}
  \PacketRow{0x00-0x10}{Selected AID}{Application Identifier read from the credential (for example from the FCI template). Present only when the length is non-zero.}{0x00-0xFF}
\end{PacketTable}
