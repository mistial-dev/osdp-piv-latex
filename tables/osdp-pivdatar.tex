  \PacketRow{1}{CMND}{Reply identifier.}{0x80}
  \PacketRow{2}{MpSizeTotal}{Length in bytes of the complete data object or tagged element, least-significant byte first.}{0x0000-0xFFFF}
  \PacketRow{2}{MpOffset}{Offset in bytes of this fragment within the complete payload, least-significant byte first.}{0x0000-0xFFFF}
  \PacketRow{2}{MpFragmentSize}{Length in bytes of the data carried in this fragment, least-significant byte first.}{0x0000-0xFFFF}
