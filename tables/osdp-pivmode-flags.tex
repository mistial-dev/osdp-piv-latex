\begin{ValueDescriptionTable}{PIV Mode Flags}{Mask}
  \label{tab:pivmode-flags}
  \ValueDescriptionRow{0x01}{Enable autonomous PIV (PIV Auto). The PD prepares the credential, derives a KMAC256 challenge from the configured counters and entropy, and reports the authentication result using \code{osdp_PIVSTATUS}.}
  \ValueDescriptionRow{0x02}{Enable credential Secure Messaging preparation over contact and contactless interfaces. On contactless, also enable VCI when the credential permits VCI without pairing. Applies to autonomous and commanded operation.}
  \ValueDescriptionRow{0x04}{Enforce FICAM strict mode (PIV applications only).}
\end{ValueDescriptionTable}
