\begin{ValueDescriptionTable}{SPE Mode Values}{Value}
  \label{tab:pivspe-modes}
  \ValueDescriptionRow{0x00}{Report the persistent SPE status without initiating PIN entry.}
  \ValueDescriptionRow{0x01}{Acquire a PIN and verify it with the credential currently present.}
  \ValueDescriptionRow{0x02}{Verify the cached PIN over established VCI.}
  \ValueDescriptionRow{0x03}{Verify the cached PIN over the selected credential interface. Use established credential SM when present.}
  \ValueDescriptionRow{0x04}{Acquire and cache a PIN.}
  \ValueDescriptionRow{0x05}{Abort active PIN entry and clear the cached PIN.}
  \ValueDescriptionRow{0x06}{Report attempts remaining without initiating PIN verification.}
  \ValueDescriptionRow{0x07-0x7F}{Reserved for future use.}
  \ValueDescriptionRow{0x80-0xFF}{Private use.}
\end{ValueDescriptionTable}
