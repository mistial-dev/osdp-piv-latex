\begin{ValueDescriptionTable}{SPE Mode Values}{Value}
  \label{tab:pivspe-modes}
  \ValueDescriptionRow{0x00}{Status only – report current SPE state without initiating entry.}
  \ValueDescriptionRow{0x01}{Normal – single PIN verification with the credential present (contact).}
  \ValueDescriptionRow{0x02}{VCI pre-cache – collect a PIN prior to credential presentation for VCI use.}
  \ValueDescriptionRow{0x03}{VCI execute – perform PIN verification over VCI using a cached PIN.}
  \ValueDescriptionRow{0x04}{PIN caching – cache PIN for subsequent operations (non-FICAM use cases).}
  \ValueDescriptionRow{0x05}{Clear/abort – clear any cached PIN and abort the current operation.}
  \ValueDescriptionRow{0x06-0x7F}{Reserved for future use (set to zero).}
  \ValueDescriptionRow{0x80-0xFF}{Private use.}
\end{ValueDescriptionTable}
