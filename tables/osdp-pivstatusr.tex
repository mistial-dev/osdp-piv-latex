\begin{PacketTable}{PIV Auto Status Reply (osdp\_PIVSTATUSR)}
  \PacketRow{1}{CMND}{Reply identifier.}{0x89 (tentative)}
  \PacketRow{2}{MpSizeTotal}{Total size of the complete PIV Auto status payload, least-significant byte first.}{0x0000--0xFFFF}
  \PacketRow{2}{MpOffset}{Offset of this fragment within the complete status payload, least-significant byte first.}{0x0000--0xFFFF}
  \PacketRow{2}{MpFragmentSize}{Length of the status payload bytes in this fragment, least-significant byte first.}{0x0000--0xFFFF}
  \PacketRow{1}{Result}{PIV Auto result for the challenged credential.}{0x00 success, other values failure}
  \PacketRow{1}{Detail}{Additional result detail. Reserved when Result is 0x00.}{0x00--0xFF}
  \PacketRow{4}{Series Counter}{Series counter value used to derive the challenge, least-significant byte first.}{0x00000000--0xFFFFFFFF}
  \PacketRow{4}{Sequence Counter}{Sequence counter value used to derive the challenge, least-significant byte first.}{0x00000000--0xFFFFFFFF}
  \PacketRow{25}{FASC-N}{FASC-N read from the credential.}{200-bit FASC-N}
  \PacketRow{16}{UUID}{Credential UUID read from the credential, or all zeroes when unavailable.}{16 octets}
  \PacketRow{1}{Key Reference}{PIV key reference challenged by the PD.}{For example 0x9E}
  \PacketRow{1}{Algorithm Identifier}{PIV GENERAL AUTHENTICATE algorithm identifier.}{For example 0x07}
  \PacketRow{2}{Signed Response Length}{Total length of the signed response, least-significant byte first.}{0x0000--0xFFFF}
  \PacketRow{0--n}{Signed Response Fragment}{Contiguous bytes of the signed response carried in this fragment.}{Variable}
\end{PacketTable}
