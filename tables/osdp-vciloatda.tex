\begin{PacketTable}{VCI Trust Anchor Load (osdp\_PIV\_VCILOADTA)}
  \PacketRow{1}{CMND}{Command identifier.}{0xAF (tentative)}
  \PacketRow{1}{Operation}{\texttt{0x00} load/replace anchor; \texttt{0x01} delete anchor; \texttt{0x02} query anchor inventory.}{0x00--0x02}
  \PacketRow{1}{Trust Anchor ID}{Controller-assigned identifier for the anchor (1--128).}{0x00--0xFF}
  \PacketRow{8}{Issuer Identifier Number}{The leftmost eight bytes of the subjectKeyIdentifier from the content-signing X.509 certificate. This value \textbf{SHALL} match the IIN field in the VCI trust anchor record (tag \texttt{0x42}) and is used as the anchor selector during VCI establishment.}{0x00--0xFF}
  \PacketRow{2}{Anchor Length}{Total size of the variable-length VCI trust anchor record, least-significant byte first. Set to \texttt{0x0000} when deleting or querying inventory.}{0x0000--0x0240}
  \PacketRow{2}{MpSizeTotal}{Total size of the complete anchor data, least-significant byte first. For load operations, this value equals \texttt{Anchor Length}. Set to \texttt{0x0000} when deleting or querying inventory.}{0x0000--0x0240}
  \PacketRow{2}{MpOffset}{Offset of this fragment within the anchor data, least-significant byte first. Set to \texttt{0x0000} when deleting or querying inventory.}{0x0000--0x023F}
  \PacketRow{2}{MpFragmentSize}{Length of the data block in this fragment, least-significant byte first. \texttt{MpOffset} + \texttt{MpFragmentSize} \textbf{SHALL NOT} exceed \texttt{Anchor Length}. Set to \texttt{0x0000} when deleting or querying inventory.}{0x0000--0x0240}
  \PacketRow{0--n}{Anchor Data}{Fragment bytes of the OSDP-defined VCI trust anchor record. Omitted for delete and query operations.}{0x00--0xFF}
\end{PacketTable}
